cmake_minimum_required(VERSION 3.22)
project(chinesechess_ai_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(AI_ENGINE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../Pikafish/source-m4/Pikafish-Pikafish-2026-09-06/src" CACHE PATH "Pinned AI engine source directory")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py"
  "${AI_ENGINE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/upstream" COMMAND_ERROR_IS_FATAL ANY)
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/upstream/*.cpp")
list(FILTER ENGINE_SOURCES EXCLUDE REGEX "/main.cpp$|/universal/")
add_library(ai_engine ${ENGINE_SOURCES} src/ai_engine.cpp)
target_include_directories(ai_engine PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>"
  PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/upstream")
target_compile_definitions(ai_engine PRIVATE NDEBUG ZSTD_DISABLE_ASM=1 PF_BUILD)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  target_compile_definitions(ai_engine PRIVATE IS_64BIT)
else()
  message(FATAL_ERROR "AI library supports only 64-bit targets")
endif()
if(CMAKE_OSX_ARCHITECTURES STREQUAL "arm64" OR ANDROID_ABI STREQUAL "arm64-v8a")
  target_compile_definitions(ai_engine PRIVATE USE_NEON=8)
elseif(CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64" OR ANDROID_ABI STREQUAL "x86_64" OR WIN32)
  target_compile_definitions(ai_engine PRIVATE USE_SSE2)
endif()
if(BUILD_SHARED_LIBS)
  target_compile_definitions(ai_engine PUBLIC PF_SHARED)
endif()
if(MSVC)
  target_compile_options(ai_engine PRIVATE /O2 /EHsc)
else()
  target_compile_options(ai_engine PRIVATE -O3 -fvisibility=hidden)
endif()
if(ANDROID)
  target_link_options(ai_engine PRIVATE "-Wl,-z,max-page-size=16384")
endif()
find_package(Threads REQUIRED)
target_link_libraries(ai_engine PRIVATE Threads::Threads)
install(TARGETS ai_engine ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/ai_engine.h DESTINATION include)
