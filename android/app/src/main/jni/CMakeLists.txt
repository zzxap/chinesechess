cmake_minimum_required(VERSION 3.22)
project(appmodules)
include(${REACT_ANDROID_DIR}/cmake-utils/ReactNative-application.cmake)
get_filename_component(CHESS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(AI_LIBRARY "${CHESS_ROOT}/AI/lib-dist/android/jniLibs/${ANDROID_ABI}/libai_engine.so")
if(NOT EXISTS "${AI_LIBRARY}")
  message(FATAL_ERROR "Bundled AI library missing; build-ai-engine-libs.sh android first")
endif()
add_library(ai_engine SHARED IMPORTED)
set_target_properties(ai_engine PROPERTIES IMPORTED_LOCATION "${AI_LIBRARY}")
add_library(chessai_jni SHARED "${CHESS_ROOT}/native/ai-engine/android/chessai_jni.cpp")
target_compile_features(chessai_jni PRIVATE cxx_std_17)
target_link_libraries(chessai_jni PRIVATE ai_engine)
target_link_options(chessai_jni PRIVATE "-Wl,-z,max-page-size=16384")
