cmake_minimum_required(VERSION 3.22)
project(chinesechess_pikafish LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(PIKAFISH_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../Pikafish/source-m4/Pikafish-Pikafish-2026-09-06/src" CACHE PATH "Pinned Pikafish source directory")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py"
  "${PIKAFISH_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/upstream" COMMAND_ERROR_IS_FATAL ANY)
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/upstream/*.cpp")
list(FILTER ENGINE_SOURCES EXCLUDE REGEX "/main.cpp$|/universal/")
add_library(pikafish ${ENGINE_SOURCES} src/pikafish.cpp)
target_include_directories(pikafish PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>"
  PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/upstream")
target_compile_definitions(pikafish PRIVATE NDEBUG ZSTD_DISABLE_ASM=1 PF_BUILD)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  target_compile_definitions(pikafish PRIVATE IS_64BIT)
else()
  message(FATAL_ERROR "Pikafish library supports only 64-bit targets")
endif()
if(CMAKE_OSX_ARCHITECTURES STREQUAL "arm64" OR ANDROID_ABI STREQUAL "arm64-v8a")
  target_compile_definitions(pikafish PRIVATE USE_NEON=8)
elseif(CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64" OR ANDROID_ABI STREQUAL "x86_64" OR WIN32)
  target_compile_definitions(pikafish PRIVATE USE_SSE2)
endif()
if(BUILD_SHARED_LIBS)
  target_compile_definitions(pikafish PUBLIC PF_SHARED)
endif()
if(MSVC)
  target_compile_options(pikafish PRIVATE /O2 /EHsc)
else()
  target_compile_options(pikafish PRIVATE -O3 -fvisibility=hidden)
endif()
if(ANDROID)
  target_link_options(pikafish PRIVATE "-Wl,-z,max-page-size=16384")
endif()
find_package(Threads REQUIRED)
target_link_libraries(pikafish PRIVATE Threads::Threads)
install(TARGETS pikafish ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/pikafish.h DESTINATION include)
